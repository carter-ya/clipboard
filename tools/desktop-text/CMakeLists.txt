cmake_minimum_required(VERSION 3.22)
project(ClipboardTextWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
endforeach()
find_package(Threads REQUIRED)
file(GLOB FASTTEXT_SOURCES "${FASTTEXT_SOURCE}/src/*.cc")
list(FILTER FASTTEXT_SOURCES EXCLUDE REGEX "/main\\.cc$")
add_library(clipboard-fasttext STATIC ${FASTTEXT_SOURCES})
target_include_directories(clipboard-fasttext PUBLIC "${FASTTEXT_SOURCE}/src")
target_link_libraries(clipboard-fasttext PRIVATE Threads::Threads)
add_library(clipboard-unicode STATIC "${UNICODE_SOURCE}/src/unicode.cpp" "${UNICODE_SOURCE}/src/unicode-data.cpp")
target_include_directories(clipboard-unicode PUBLIC "${UNICODE_SOURCE}/src")
add_executable(clipboard-text-worker text-worker.cpp "${UNICODE_SOURCE}/vendor/hash/sha256/sha256.c")
target_include_directories(clipboard-text-worker PRIVATE "${ONNX_INCLUDE}" "${GENERATED_INCLUDE}"
  "${UNICODE_SOURCE}/vendor/nlohmann" "${UNICODE_SOURCE}/vendor/hash" "${UNICODE_SOURCE}/vendor/hash/sha256")
target_link_libraries(clipboard-text-worker PRIVATE clipboard-fasttext clipboard-unicode "${ONNX_LIBRARY}")
if(MSVC)
  target_compile_options(clipboard-text-worker PRIVATE /utf-8)
  target_compile_definitions(clipboard-fasttext PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
  target_link_options(clipboard-text-worker PRIVATE "LINKER:/SUBSYSTEM:WINDOWS" "LINKER:/ENTRY:mainCRTStartup")
else()
  # Installed helpers cannot silently fall back to a development SDK directory.
  set_target_properties(clipboard-text-worker PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
add_executable(tokenizer-reference-check tokenizer-probe.cpp)
target_include_directories(tokenizer-reference-check PRIVATE "${UNICODE_SOURCE}/vendor/nlohmann")
target_link_libraries(tokenizer-reference-check PRIVATE clipboard-unicode)
