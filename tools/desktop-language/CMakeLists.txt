cmake_minimum_required(VERSION 3.22)
project(ClipboardLanguageWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
endforeach()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_CONNECT OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
foreach(backend CUDA HIP MUSA METAL VULKAN SYCL RPC OPENCL CANN ZENDNN WEBGPU)
  set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature AVX AVX2 AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AVX_VNNI BMI2 FMA F16C SSE42 AMX_TILE AMX_INT8 AMX_BF16)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
  foreach(feature AVX AVX2 FMA F16C)
    set(GGML_${feature} ON CACHE BOOL "" FORCE)
  endforeach()
endif()
add_subdirectory("${LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_executable(language-cpu-check cpu-check.cpp)
add_executable(clipboard-language-worker language-worker.cpp "${LLAMA_SOURCE}/vendor/hash/sha256/sha256.c")
target_include_directories(clipboard-language-worker PRIVATE
  "${LLAMA_SOURCE}/vendor/nlohmann" "${LLAMA_SOURCE}/vendor/hash"
  "${LLAMA_SOURCE}/vendor/hash/sha256" "${MODEL_INCLUDE}")
target_link_libraries(clipboard-language-worker PRIVATE llama)
if(MSVC)
  target_compile_options(clipboard-language-worker PRIVATE /utf-8)
endif()
if(WIN32)
  target_compile_definitions(clipboard-language-worker PRIVATE NOMINMAX)
  target_link_options(clipboard-language-worker PRIVATE "LINKER:/SUBSYSTEM:WINDOWS" "LINKER:/ENTRY:mainCRTStartup")
endif()
